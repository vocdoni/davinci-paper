% Make section entries bold in ToC
\makeatletter
\renewcommand{\@dotsep}{4.5}
\renewcommand{\l@section}[2]{\addpenalty{\@secpenalty}
	\addvspace{1.0em plus 1pt}
	\@tempdima=1.5em
	\parindent \z@ \rightskip \@pnumwidth
	\parfillskip -\@pnumwidth
	{\leavevmode
		\advance\leftskip\@tempdima
		\hskip -\leftskip
		\bfseries #1\hfil \hbox to\@pnumwidth{\hss #2}}\par}
\makeatother

% Adds vertical space before and after \subsubsection headings and ensures body text starts on a new line
\makeatletter

	{-18\p@ \@plus -4\p@ \@minus -4\p@}%   % Before skip
	{4\p@ \@plus 2\p@ \@minus 2\p@}%      % After skip
	{\normalfont\normalsize\bfseries}}   % Font style
\makeatother
